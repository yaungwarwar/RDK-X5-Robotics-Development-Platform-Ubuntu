```bash

import json, subprocess, time, urllib.request
unit='rdk-control-center.service'
def cmd(args):
    p=subprocess.run(args, text=True, stdout=subprocess.PIPE, stderr=subprocess.STDOUT, timeout=10)
    return p.stdout.strip() if p.returncode == 0 else 'ERROR '+p.stdout.strip()
def get(url):
    try:
        with urllib.request.urlopen(url, timeout=5) as r: return json.load(r)
    except Exception as e: return {'error':str(e)}
start=time.monotonic(); prev=None
for i in range(31):
    target=start+10*i
    time.sleep(max(0,target-time.monotonic()))
    now=time.monotonic()
    raw=cmd(['systemctl','show',unit,'-p','CPUUsageNSec','--value'])
    try: cpu=int(raw)
    except ValueError: cpu=None
    pct=None if prev is None or cpu is None or prev[1] is None else round((cpu-prev[1])/((now-prev[0])*1e9)*100,3)
    prev=(now,cpu)
    print('\nSAMPLE',i,'elapsed',round(now-start,2),'backend CPU % one core',pct,flush=True)
    print(cmd(['systemctl','show',unit,'-p','MainPID','-p','MemoryCurrent','-p','TasksCurrent']))

    print('ratio',cmd(['cat','/sys/devices/platform/soc/3a000000.bpu/ratio']))
    print('processes',cmd(['ps','-eo','pid,ppid,stat,args']))
    print('',flush=True)
PY
```
